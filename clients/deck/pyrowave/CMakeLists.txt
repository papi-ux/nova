cmake_minimum_required(VERSION 3.20)
project(nova_pyrowave LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PYROWAVE REQUIRED IMPORTED_TARGET pyrowave-shared=0.6.0 vulkan)
add_library(nova_pyrowave STATIC codec.cpp)
target_include_directories(nova_pyrowave PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nova_pyrowave PUBLIC PkgConfig::PYROWAVE)
target_compile_options(nova_pyrowave PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
    add_executable(nova_pyrowave_protocol_test protocol_test.cpp)
    add_test(NAME nova_pyrowave_protocol_test COMMAND nova_pyrowave_protocol_test)
    add_executable(nova_pyrowave_parser_test parser_test.cpp)
    target_link_libraries(nova_pyrowave_parser_test PRIVATE nova_pyrowave)
    add_test(NAME nova_pyrowave_parser_test COMMAND nova_pyrowave_parser_test)
    set_tests_properties(nova_pyrowave_parser_test PROPERTIES TIMEOUT 30)
    add_executable(nova_pyrowave_test codec_test.cpp)
    target_link_libraries(nova_pyrowave_test PRIVATE nova_pyrowave)
    add_test(NAME nova_pyrowave_test COMMAND nova_pyrowave_test)
    set_tests_properties(nova_pyrowave_test PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
endif()
